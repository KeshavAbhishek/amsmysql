        function deleteMode() {
            var hideThese = document.querySelectorAll(".hide");
            for (const x of hideThese) {
                x.style.display = "none";
            }
            document.getElementById("form").setAttribute("action", "./delete.php");
            document.getElementById("submit").innerText = "Delete Ticket";

            document.getElementById("verticalNav").style.height = "0rem";
            open = 0;

            document.getElementById("head").innerHTML = "Delete<br>Customer Details";

        }

        // Update Function

        function updateMode() {
            var hideThese = document.querySelectorAll(".hide");
            for (const x of hideThese) {
                x.style.display = "none";
            }
            document.getElementById("form").setAttribute("action", "./update.php");
            document.getElementById("submit").innerText = "Proceed";

            document.getElementById("verticalNav").style.height = "0rem";
            open = 0;

            document.getElementById("head").innerHTML = "Update<br>Customer Details";
        }

        document.getElementById("iTag").addEventListener("click", () => {
            if (open == 0) {
                document.getElementById("verticalNav").style.height = "5rem";
                open = 1;
            }
            else {
                document.getElementById("verticalNav").style.height = "0rem";
                open = 0;
            }
        });
